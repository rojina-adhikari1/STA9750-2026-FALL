# Same caching idea as MP02: only hit the API if the file isn't saved yet.
# To refresh (e.g. after adding variables), delete the matching .rds file.
get_acs_cached <- function(file_name, ...){
    file_path <- fs::path(DATA_DIR, paste0(file_name, ".rds"))

    if(!fs::file_exists(file_path)){
        result <- get_acs(...)
        saveRDS(result, file_path)
    }

    readRDS(file_path)
}

# Task 3: download, unzip and read the 2020 NTA shapefile (not via tidycensus)
load_nyc_nta <- function(){
    NTA_URL  <- "https://s-media.nyc.gov/agencies/dcp/assets/files/zip/data-tools/bytes/neighborhood-tabulation-areas/nynta2020_26c.zip"
    zip_file <- fs::path(DATA_DIR, "nynta2020.zip")

    if(!fs::file_exists(zip_file)){
        request(NTA_URL) |>
            req_headers(
                `User-Agent` = "Mozilla/5.0 (Macintosh; Intel Mac OS X 10.15; rv:149.0) Gecko/20100101 Firefox/149.0"
            ) |>
            req_retry(max_tries = 4) |>
            req_perform(path = zip_file)
    }

    # Stop early if the download isn't a real zip (e.g. a web page)
    # (every real zip file starts with the letters "PK")
    zip_ok <- identical(readBin(zip_file, "raw", 2), charToRaw("PK"))
    if(!zip_ok){
        fs::file_delete(zip_file)
        stop("NTA download is not a valid zip file. Check NTA_URL.")
    }

    shp_file <- fs::dir_ls(DATA_DIR, recurse = TRUE, glob = "*nynta2020*.shp")

    if(length(shp_file) == 0){
        unzip(zip_file, exdir = DATA_DIR, unzip = "unzip")  # macOS system unzip
        shp_file <- fs::dir_ls(DATA_DIR, recurse = TRUE, glob = "*nynta2020*.shp")
    }

    st_read(shp_file[1], quiet = TRUE) |>
        # NTAType: 0 = residential neighborhood; other codes are parks,
        # airports, cemeteries etc. Kept so those can be filtered out later.
        select(BoroName, NTA2020, NTAName, NTAType)
}

nyc_nta <- load_nyc_nta()

# Task 5: page through the PLUTO API and save one CSV (cached like MP02)
load_pluto <- function(){
    PLUTO_FILE <- fs::path(DATA_DIR, "pluto.csv")

    if(!fs::file_exists(PLUTO_FILE)){
        BASE_URL  <- "https://data.cityofnewyork.us/resource/64uk-42ks.csv"
        # Only the columns needed later
        PLUTO_COLS <- c("bbl", "borough", "bct2020", "landuse",
                        "lotarea", "bldgarea", "comarea", "resarea",
                        "officearea", "retailarea",
                        "unitsres", "unitstotal", "numfloors")
        PAGE_SIZE <- 50000
        offset    <- 0
        pages     <- list()

        repeat {
            page <- request(BASE_URL) |>
                req_url_query(
                    `$select` = paste(PLUTO_COLS, collapse = ","),
                    `$order`  = "bbl",          # fixed order so pages don't overlap
                    `$limit`  = PAGE_SIZE,
                    `$offset` = offset
                ) |>
                req_retry(max_tries = 4) |>
                req_perform() |>
                resp_body_string() |>
                read_csv(col_types = cols(.default = col_character()))

            if(nrow(page) == 0) break
            pages[[length(pages) + 1]] <- page
            if(nrow(page) < PAGE_SIZE) break
            offset <- offset + PAGE_SIZE
        }

        bind_rows(pages) |> write_csv(PLUTO_FILE)
    }

    read_csv(
        PLUTO_FILE,
        col_types = cols(
            bbl = col_character(), borough = col_character(),
            bct2020 = col_character(), landuse = col_character(),
            .default = col_double()
        )
    )
}

pluto <- load_pluto()

# bct2020 = borough digit + 6-digit tract (e.g. 1000100).
# Convert to the Census GEOID (e.g. 36061000100) so PLUTO can join to tracts.
BORO_TO_COUNTY <- c("1" = "061", "2" = "005", "3" = "047", "4" = "081", "5" = "085")

pluto <- pluto |>
    mutate(
        bct2020 = str_remove(bct2020, "\\.0+$"),
        GEOID   = paste0("36",
                         BORO_TO_COUNTY[str_sub(bct2020, 1, 1)],
                         str_sub(bct2020, 2))
    )

# One row per tract with building totals, for the EDA questions
tract_buildings <- pluto |>
    filter(!is.na(bct2020)) |>
    group_by(GEOID) |>
    summarize(
        office_area = sum(officearea, na.rm = TRUE),
        res_area    = sum(resarea,    na.rm = TRUE),
        com_area    = sum(comarea,    na.rm = TRUE),
        bldg_area   = sum(bldgarea,   na.rm = TRUE),
        res_units   = sum(unitsres,   na.rm = TRUE),
        max_floors  = if (all(is.na(numfloors))) NA_real_ else max(numfloors, na.rm = TRUE),
        .groups     = "drop"
    )

# Small denominators give silly results (e.g. a tract with 2 apartments and

q3 <- tract_full |>

# Turn labels like "Estimate!!Total Groups Tallied:!!South Asian:!!Nepalese"
# into a broad region ("South Asian") and keep only the bottom-level rows,
# so nothing is counted twice.
b02018_labels <- acs1_vars |>
    filter(str_starts(name, "B02018_")) |>
    transmute(
        variable = name,
        depth    = str_count(label, "!!"),
        region   = str_remove(str_split_i(label, "!!", 3), ":$"),
        is_header = str_ends(label, ":")
    ) |>
    filter(depth >= 2, !is_header)

q5 <- cbsa_asian_groups |>
    filter(NAME == COMPARISON_METRO) |>
    inner_join(b02018_labels, by = "variable") |>
    group_by(region) |>
    summarize(people = sum(estimate, na.rm = TRUE), .groups = "drop")

ggplot(q5, aes(x = people, y = fct_reorder(region, people))) +
    geom_col(fill = "#2a7f8f") +
    scale_x_continuous(labels = comma) +

        title    = paste("Asian Ancestry Groups in", COMPARISON_METRO),
